```python file=biz/expense/calc.py
from decimal import Decimal, InvalidOperation
from typing import Iterable, Optional, Union

from biz.expense.models import Expense


def top_category(expenses: Iterable[Expense]) -> Optional[str]:
    """金额最高的类别；没有数据时返回 None。"""
    totals = summarize(expenses)
    if not totals:
        return None
    return max(totals.items(), key=lambda kv: kv[1])[0]


def summarize(
    expenses: Iterable[Expense],
    min_amount: Optional[Union[Decimal, str, int, float]] = None,
) -> dict[str, Decimal]:
    """按类别汇总金额，可按最低金额阈值过滤；金额为 Decimal。"""
    threshold: Optional[Decimal] = None
    if min_amount is not None:
        try:
            threshold = (
                min_amount
                if isinstance(min_amount, Decimal)
                else Decimal(str(min_amount))
            )
        except (InvalidOperation, TypeError, ValueError) as exc:
            raise ValueError("min_amount must be a finite non-negative number") from exc

        if not threshold.is_finite():
            raise ValueError("min_amount must be finite")
        if threshold < Decimal("0"):
            raise ValueError("min_amount must be non-negative")

    totals: dict[str, Decimal] = {}
    for e in expenses:
        if threshold is not None and e.amount < threshold:
            continue
        totals[e.category] = totals.get(e.category, Decimal("0")) + e.amount
    return dict(sorted(totals.items()))


def total_amount(expenses: Iterable[Expense]) -> Decimal:
    return sum((e.amount for e in expenses), Decimal("0"))
```
